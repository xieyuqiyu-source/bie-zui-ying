# 数据库设计草案

## users

```sql
create table users (
  id varchar(32) primary key,
  openid varchar(128) unique,
  unionid varchar(128),
  nickname varchar(64) not null,
  avatar_url text,
  title varchar(64) not null default '神秘手指',
  total_matches int not null default 0,
  wins int not null default 0,
  losses int not null default 0,
  draws int not null default 0,
  current_streak int not null default 0,
  best_streak int not null default 0,
  best_score int not null default 0,
  created_at timestamp not null,
  updated_at timestamp not null
);
```

## rooms

```sql
create table rooms (
  id varchar(32) primary key,
  code varchar(16) not null,
  owner_id varchar(32) not null,
  guest_id varchar(32),
  status varchar(32) not null,
  owner_ready boolean not null default false,
  guest_ready boolean not null default false,
  match_id varchar(32),
  created_at timestamp not null,
  expires_at timestamp not null,
  started_at timestamp,
  ended_at timestamp
);
```

## matches

```sql
create table matches (
  id varchar(32) primary key,
  room_id varchar(32) not null,
  player_a_id varchar(32) not null,
  player_b_id varchar(32) not null,
  player_a_score int not null,
  player_b_score int not null,
  winner_id varchar(32),
  result_type varchar(16) not null,
  duration_seconds int not null,
  trash_talk text not null,
  abnormal_flag boolean not null default false,
  created_at timestamp not null
);
```

## trash_talks

```sql
create table trash_talks (
  id varchar(32) primary key,
  scene varchar(32) not null,
  content text not null,
  weight int not null default 1,
  enabled boolean not null default true,
  created_at timestamp not null,
  updated_at timestamp not null
);
```

## events

```sql
create table events (
  id varchar(32) primary key,
  user_id varchar(32),
  anonymous_id varchar(64),
  event_name varchar(64) not null,
  event_payload json,
  created_at timestamp not null
);
```
